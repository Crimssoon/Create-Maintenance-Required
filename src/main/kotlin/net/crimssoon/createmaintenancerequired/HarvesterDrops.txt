package net.crimssoon.createmaintenancerequired

import com.simibubi.create.AllBlocks
import net.minecraft.server.level.ServerLevel
import net.neoforged.bus.api.SubscribeEvent
import net.neoforged.fml.common.EventBusSubscriber
import net.neoforged.neoforge.event.level.BlockDropsEvent

@EventBusSubscriber(modid = CreateMaintenanceRequired.MOD_ID)
object HarvesterDrops {

    @JvmStatic
    @SubscribeEvent
    fun onBlockDrops(event: BlockDropsEvent) {
        if (event.level !is ServerLevel) return
        if (event.state.block != AllBlocks.MECHANICAL_HARVESTER.get()) return

        val be = event.blockEntity ?: return
        val durability = HarvesterDurability.get(be)
        if (durability >= HarvesterDurability.MAX) return

        val harvesterItem = AllBlocks.MECHANICAL_HARVESTER.asItem()
        for (drop in event.drops) {
            if (drop.item.item == harvesterItem) {
                drop.item.set(ModComponents.DURABILITY.get(), durability)
            }
        }
    }
}
