package net.crimssoon.createmaintenancerequired

import com.simibubi.create.content.contraptions.actors.harvester.HarvesterBlockEntity
import net.minecraft.world.level.Level
import java.util.Collections
import java.util.WeakHashMap

object HarvesterTracker {

    private val all: MutableSet<HarvesterBlockEntity> =
        Collections.newSetFromMap(WeakHashMap())

    fun add(be: HarvesterBlockEntity) {
        synchronized(all) { all.add(be) }
    }

    fun collect(level: Level, out: MutableList<HarvesterBlockEntity>) {
        synchronized(all) {
            val iterator = all.iterator()
            while (iterator.hasNext()) {
                val be = iterator.next()
                val beLevel = be.level
                if (be.isRemoved || (beLevel != null && !beLevel.isClientSide)) {
                    iterator.remove()
                    continue
                }
                if (beLevel === level) out.add(be)
            }
        }
    }
}
